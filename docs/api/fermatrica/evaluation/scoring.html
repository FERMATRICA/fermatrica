 28def scoring(ds: pd.DataFrame
 29            , pred: pd.Series | np.ndarray
 30            , model: "Model"):
 31    """
 32    Main function to create complex scoring from the set of individual scorings / metrics
 33
 34    :param ds: dataset containing Y (observed) variable
 35    :param pred: vector with predicted values
 36    :param model: FERMATRICA Model object
 37    :return:
 38    """
 39
 40    obs = ds[model.conf.Y_var]
 41    scoring_dict = copy.deepcopy(model.conf.scoring_dict)
 42
 43    # iterate over keys in scoring_dict
 44    for key in scoring_dict:
 45
 46        fn = globals()[key]
 47
 48        if ('ds' in inspect.signature(fn).parameters.keys()) and ('model' in inspect.signature(fn).parameters.keys()):
 49            score = fn(obs, pred, ds, model)
 50        elif (len(inspect.signature(fn).parameters.keys()) == 1) and ('model' in inspect.signature(fn).parameters.keys()):
 51            score = fn(model)
 52        elif 'ds' in inspect.signature(fn).parameters.keys():
 53            score = fn(obs, pred, ds)
 54        else:
 55            score = fn(obs, pred)
 56
 57        if (not hasattr(model.conf, 'custom_scoring')) or (model.conf.custom_scoring == ''):
 58
 59            # Calculate mapping of the score
 60            score = bell_curve(score, scoring_dict[key]['if_invert'], scoring_dict[key]['width'])
 61
 62            if (isinstance(score, list)) or (isinstance(score, np.ndarray)):
 63                score = np.average(score)
 64
 65        scoring_dict[key]['value'] = score
 66
 67    # calculate final value
 68
 69    if (not hasattr(model.conf, 'custom_scoring')) or (model.conf.custom_scoring == ''):
 70        rtrn = sum(scoring_dict[k]['value'] * scoring_dict[k]['weight'] for k in scoring_dict)
 71
 72    else:
 73        try:
 74            rtrn = eval(model.conf.custom_scoring_compiled)
 75        except (RuntimeError, TypeError, NameError):
 76            fermatrica_error('Custom scoring function failed. Check if all variables are set ' +
 77                         '`if_active` = 1 in the `scoring` sheet of the config / model definition file.')
 78
 79    return rtrn

 82def bell_curve(x: list | np.ndarray
 83               , if_invert: int = 0
 84               , width: float = 1):
 85    """
 86    Function to normalise raw individual scores / metrics to 0...1 range using bell curve
 87
 88    :param x:
 89    :param if_invert: Should we invert parameter? I.e. RMSE - the smaller, the better, but combined score is maximising
 90        1 = yes
 91        0 = no
 92    :param width:
 93    :return:
 94    """
 95
 96    if isinstance(x, list):
 97        x = np.array(x)
 98
 99    # function requires inversion of if_invert (ha-ha), so 0 -> 1 and all other values -> 0
100    if_invert = np.where(np.abs(if_invert) > 0, 0, 1)
101
102    rtrn = np.exp(-(x - if_invert) ** 2 / 2 * (1 / width) ** 2)
103
104    return rtrn

203def sign_simple(model: "Model") -> int | float:
204    """
205    Check if all regression coefficient signs are correct (correspond to RHS definition in Model).
206    Wrong signs are penalised with 1 * sign_weight for every occurrence, result is weighted by the
207    number of regressors.
208
209    This function doesn't take into account how big "error" is, for more elaborate approach try
210    `sign_t_value` and `sign_t_value_s`.
211
212    :param model: FERMATRICA Model object
213    :return:
214    """
215
216    model_rhs = model.conf.model_rhs[['token', 'sign', 'sign_weight']].copy()
217    model_rhs['token'] = model_rhs['token'].apply(lambda x: re.sub(r' : ', ':', x))
218    # maybe squeeze all spaces - check later
219
220    rtrn = pd.merge(model.obj.models['main'].tvalues.to_frame('t_value').reset_index().rename(columns={'index': 'token'})
221                    , model_rhs
222                    , how='inner', on=['token'], sort=False)
223
224    rtrn = np.sign(rtrn['t_value']) * rtrn['sign'] * rtrn['sign_weight']
225    rtrn = np.abs(rtrn[rtrn < 0].sum()) / model.obj.models['main'].tvalues.shape[0]
226
227    return rtrn

230def sign_t_value(model: "Model") -> float | int:
231    """
232    Check if all regression coefficient signs are correct (correspond to RHS definition in Model).
233    Wrong signs are penalised with T-value * sign_weight for every occurrence, result is weighted by the
234    number of regressors.
235
236    T-value allows to measure how big the "error" is, so facilitates algorithmic optimisation greatly.
237    T-value is preferred over p-value due to the already existing sign and range [-Inf, Inf] rather
238    than [0, Inf].
239
240    For simpler approach see `sign_simple`, for even more sophisticated - `sign_t_value_s`
241
242    :param model: FERMATRICA Model object
243    :return:
244    """
245
246    model_rhs = model.conf.model_rhs[['token', 'sign', 'sign_weight']].copy()
247    model_rhs['token'] = model_rhs['token'].apply(lambda x: re.sub(r' : ', ':', x))
248    # maybe squeeze all spaces - check later
249
250    rtrn = pd.merge(model.obj.models['main'].tvalues.to_frame('t_value').reset_index().rename(columns={'index': 'token'})
251                    , model_rhs
252                    , how='inner', on=['token'], sort=False)
253
254    rtrn = rtrn['t_value'] * rtrn['sign'] * rtrn['sign_weight']
255    rtrn = np.abs(rtrn[rtrn < 0].sum()) / model.obj.models['main'].tvalues.shape[0]
256
257    return rtrn

260def sign_t_value_s(model: "Model") -> float | int:
261    """
262    Check if all regression coefficient signs are correct (correspond to RHS definition in Model) and
263    estimations are significant. Wrong signs and not enough significance are penalised with
264    (T-value - 2) * sign_weight for every occurrence, result is weighted by the number of regressors.
265
266    T-value allows to measure how big the "error" is, so facilitates algorithmic optimisation greatly.
267    T-value is preferred over p-value due to the already existing sign and range [-Inf, Inf] rather
268    than [0, Inf].
269
270    For simpler approaches see `sign_simple` and `sign_t_value_s`, for more flexible - `sign_t_value_st`
271
272    :param model: FERMATRICA Model object
273    :return:
274    """
275
276    model_rhs = model.conf.model_rhs[['token', 'sign', 'sign_weight']].copy()
277    model_rhs['token'] = model_rhs['token'].apply(lambda x: re.sub(r' : ', ':', x))
278    # maybe squeeze all spaces - check later
279
280    rtrn = pd.merge(model.obj.models['main'].tvalues.to_frame('t_value').reset_index().rename(columns={'index': 'token'})
281                    , model_rhs
282                    , how='inner', on=['token'], sort=False)
283
284    rtrn = (rtrn['t_value'] * rtrn['sign'] - 2.0) * rtrn['sign_weight']
285    rtrn = np.abs(rtrn[rtrn < 0].sum()) / model.obj.models['main'].tvalues.shape[0]
286
287    return rtrn

290def sign_t_value_st(model: "Model") -> float | int:
291    """
292    Check if all regression coefficient signs are correct (correspond to RHS definition in Model) and
293    estimations are significant. Wrong signs and not enough significance are penalised with
294    (T-value - threshold) * sign_weight for every occurrence, result is weighted by the number of regressors.
295
296    T-value allows to measure how big the "error" is, so facilitates algorithmic optimisation greatly.
297    T-value is preferred over p-value due to the already existing sign and range [-Inf, Inf] rather
298    than [0, Inf].
299
300    Thresholds should be defined in Model.model_rhs DataFrame in `signif_threshold` column.
301
302    For less flexible approach (threshold is allways equal to 2) see `sign_t_value_s`
303
304    :param model: FERMATRICA Model object
305    :return:
306    """
307
308    model_rhs = model.conf.model_rhs[['token', 'sign', 'sign_weight', 'signif_threshold']].copy()
309    model_rhs['token'] = model_rhs['token'].apply(lambda x: re.sub(r' : ', ':', x))
310    # maybe squeeze all spaces - check later
311
312    rtrn = pd.merge(model.obj.models['main'].tvalues.to_frame('t_value').reset_index().rename(columns={'index': 'token'})
313                    , model_rhs
314                    , how='inner', on=['token'], sort=False)
315
316    rtrn = (rtrn['t_value'] * rtrn['sign'] - rtrn['signif_threshold']) * rtrn['sign_weight']
317    rtrn = np.abs(rtrn[rtrn < 0].sum()) / model.obj.models['main'].tvalues.shape[0]
318
319    return rtrn

322def t_value(model: "Model") -> float | int:
323    """
324    Check if all regression coefficient estimations are significant by t-values.
325    Not enough significance is penalised with ABS(T-value * signif_weight) for every occurrence,
326    result is weighted by the number of regressors.
327
328    T-value allows to measure how big the "error" is, so facilitates algorithmic optimisation greatly.
329    T-value is preferred over p-value due to the already existing sign and range [-Inf, Inf] rather
330    than [0, Inf].
331
332    :param model: FERMATRICA Model object
333    :return:
334    """
335
336    model_rhs = model.conf.model_rhs[['token', 'signif_weight']].copy()
337    model_rhs['token'] = model_rhs['token'].apply(lambda x: re.sub(r' : ', ':', x))
338    # maybe squeeze all spaces - check later
339
340    rtrn = pd.merge(model.obj.models['main'].tvalues.to_frame('t_value').reset_index().rename(columns={'index': 'token'})
341                    , model_rhs
342                    , how='inner', on=['token'], sort=False)
343
344    rtrn = rtrn['t_value'] * rtrn['signif_weight']
345    rtrn = rtrn.abs().sum() / model.obj.models['main'].tvalues.shape[0]
346
347    return rtrn

350def p_value(model: "Model") -> float | int:
351    """
352    Check if all regression coefficient estimations are significant by p-values.
353    Not enough significance to be calculated as (p-value * signif_weight) for every occurrence,
354    result is weighted by the number of regressors.
355
356    IMPORTANT. In this function the smaller is the result, the better.
357
358    :param model: FERMATRICA Model object
359    :return:
360    """
361
362    model_rhs = model.conf.model_rhs[['token', 'signif_weight']].copy()
363    model_rhs['token'] = model_rhs['token'].apply(lambda x: re.sub(r' : ', ':', x))
364    # maybe squeeze all spaces - check later
365
366    if not hasattr(model.obj.models['main'], 'pvalues'):
367        logging.error('P-values cannot be calculated. Supported model types: OLS, LME, LMEA. ' +
368                      'Please use t-values instead or change model type to OLS, LME or LMEA')
369        return 0
370
371    rtrn = pd.merge(model.obj.models['main'].pvalues.to_frame('p_value').reset_index().rename(columns={'index': 'token'})
372                    , model_rhs
373                    , how='inner', on=['token'], sort=False)
374
375    rtrn = rtrn['p_value'] * rtrn['signif_weight']
376    rtrn = rtrn.abs().sum() / model.obj.models['main'].tvalues.shape[0]
377
378    return rtrn

381def durbin_watson_score(model: "Model"):
382    """
383    Durbin-Watson test for autocorrelation. Assumes OLS linear model as `model.obj.models['main']`.
384    Specific version used mostly for combined scoring
385
386    :param model: FERMATRICA Model object
387    :return:
388    """
389
390    if not isinstance(model.obj.models['main'].model, OLS):
391        logging.error('Durbin-Watson cannot be calculated. Supported model types: OLS. Supported data type: time series. ' +
392                      'Please change the model type to OLS and be sure data is a time series (not a panel nor slices).')
393        return 0
394
395    rtrn = abs(durbin_watson(model) - 2)
396
397    return rtrn
