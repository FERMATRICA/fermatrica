 24class ModelObj(StableClass):
 25    """
 26    ModelObj (model objects) keeps programming / callable objects of the model.
 27    Use as the component / attribute of `fermatrica.model.model.Model`
 28    """
 29
 30    transform_lhs_fn: "Callable | None"
 31    transform_lhs_src: str | None
 32    custom_predict_fn: "Callable | None"
 33    custom_predict_nm: str | None
 34    models: dict | None
 35    adhoc_code_src: dict | None
 36
 37    def __init__(
 38            self,
 39            model_conf: "ModelConf",
 40            custom_predict_fn: "Callable | None" = None,
 41            adhoc_code: list | None = None,
 42            if_stable: bool = True
 43    ):
 44        """
 45        Initialise instance
 46
 47        :param model_conf: ModelConf object (created via Model() before ModelObj)
 48        :param custom_predict_fn: function for custom prediction (mostly category model), pass if model
 49            is created from scratch
 50        :param adhoc_code: list of loaded Python modules with adhoc code, pass if model is created from scratch
 51        :param if_stable: prevent instance from creating new objects after initialising
 52        """
 53
 54        # code objects
 55        self.adhoc_code_fill(adhoc_code)
 56
 57        # dictionary for future object models
 58        self.models = {}
 59
 60        # LHS transformation function
 61        self._transform_lhs_generate(model_conf)
 62
 63        # custom prediction function
 64        self.custom_predict_fn = custom_predict_fn
 65
 66        if isinstance(self.custom_predict_fn, Callable):
 67            self.custom_predict_nm = custom_predict_fn.__module__ + '.' + custom_predict_fn.__name__
 68        else:
 69            self.custom_predict_nm = None
 70
 71        # prevent from creating new attributes
 72        # use instead of __slots__ to allow dynamic attribute creation during initialisation
 73        if if_stable:
 74            self._init_finish()
 75
 76    def _transform_lhs_generate(self
 77                                , model_conf: "ModelConf"):
 78        """
 79        Generate LHS function from LHS table in ModelConf
 80
 81        :param model_conf: ModelConf object
 82        :return:
 83        """
 84
 85        if hasattr(model_conf, 'model_lhs') and isinstance(model_conf.model_lhs, pd.DataFrame):
 86            self.transform_lhs_fn, self.transform_lhs_src = transform_lhs_generate(model_conf.model_lhs)
 87
 88    def adhoc_code_fill(self
 89                        , adhoc_code: list):
 90        """
 91        Get source code from loaded adhoc Python modules and save them as `adhoc_code_src` attribute
 92
 93        :param adhoc_code: adhoc Python modules
 94        :return:
 95        """
 96
 97        if adhoc_code is None:
 98            self.adhoc_code_src = None
 99        else:
100            self.adhoc_code_src = {}
101            for obj in adhoc_code:
102                self.adhoc_code_src[obj.__name__] = inspect.getsource(obj)
103
104    def restore_loaded(self
105                       , model_conf: "ModelConf"):
106        """
107        When pickling it is important to destroy all callable attributes as non-pickable (and potentially
108        non-restorable). When loading callable attributes to be restored from saved source code (by name
109        from saved module code or directly from saved function source code)
110
111        :param model_conf: ModelConf object
112        :return:
113        """
114
115        self._custom_predict_fn_load()
116        self._transform_lhs_generate(model_conf)
117
118    def _custom_predict_fn_load(self):
119        """
120        Restore custom predict function (mostly category model).
121        It is important cause function cannot be pickled and restored properly w/o its environment
122
123        """
124
125        if self.custom_predict_nm is not None:
126            self.custom_predict_fn = fun_find(fn_name=self.custom_predict_nm, env_code_src=self.adhoc_code_src)
127

129             , path: str = ''
130             ):
131        """
132        Save ModelObj
133
134        :param path: path to the folder
135        :return:
136        """
137
138        # path
139
140        if not os.path.isdir(path):
141            os.makedirs(path, exist_ok=True)

196def transform_lhs_generate(fn_src: pd.DataFrame | None):
197    """
198    Creates LHS transformation function from source
199
200    :param fn_src: pandas DataFrame containing order, type and source code of LHS transformations
201    :return:
202    """
203
204    if fn_src[fn_src['if_active'] == 1].shape[0] == 0:
205        # it is important to run it anyway for global optimizer
206
207        fn_src = inspect.getsourcelines(transform_lhs_empty)[0]
208        fn_src[0] = re.sub('transform_lhs_empty', 'transform_lhs_fn', fn_src[0])
209
210        fn_cl = fun_generate(fn_src=fn_src, fn_name='transform_lhs_fn')
211
212        return fn_cl, fn_src
213
214    model_lhs = fn_src[fn_src['if_active'] == 1].copy()
215
216    model_lhs_mult = model_lhs[model_lhs['type'] == "multiplicative"]
217    model_lhs_fb = model_lhs[model_lhs['type'] == "free_before"]
218    model_lhs_fa = model_lhs[model_lhs['type'] == "free_after"]
219    model_lhs_fa1x = model_lhs[model_lhs['type'] == "free_after_1x"]
220
221    fnct_src = inspect.getsourcelines(transform_lhs_template)[0]
222    fnct_src[0] = re.sub('transform_lhs_template', 'transform_lhs_fn', fnct_src[0])
223
224    indnt = [x for x in fnct_src if re.search('tmp_to_original = 1', x)][0]
225    indnt = re.search(r'^ +', indnt).group(0)
226
227    if model_lhs_fb.shape[0] > 0:
228
229        pos_cur = [i for i in range(len(fnct_src)) if re.search('tmp_to_original = 1', fnct_src[i])][0]
230        for ind, row in model_lhs_fb.iterrows():
231            fnct_src.insert(pos_cur, indnt + 'y = ' + row['token'] + '\n')
232
233        pos_cur = [i for i in range(len(fnct_src)) if re.search('tmp_from_original = 1', fnct_src[i])][0] + 1
234        for ind, row in model_lhs_fb.iterrows():
235            fnct_src.insert(pos_cur, indnt + 'y = ' + row['token'] + '\n')
236
237    if model_lhs_mult.shape[0] > 0:
238
239        pos_cur = [i for i in range(len(fnct_src)) if re.search('tmp_to_original = 1', fnct_src[i])][0]
240        for ind, row in model_lhs_mult.iterrows():
241            fnct_src.insert(pos_cur, indnt + 'y = y / (' + row['token'] + ')\n')
242
243        pos_cur = [i for i in range(len(fnct_src)) if re.search('tmp_from_original = 1', fnct_src[i])][0] + 1
244        for ind, row in model_lhs_mult.iterrows():
245            fnct_src.insert(pos_cur, indnt + 'y = y * (' + row['token'] + ')\n')
246
247    if model_lhs_fa.shape[0] > 0:
248
249        pos_cur = [i for i in range(len(fnct_src)) if re.search('tmp_to_original = 1', fnct_src[i])][0]
250        for ind, row in model_lhs_fa.iterrows():
251            fnct_src.insert(pos_cur, indnt + 'y = ' + row['token'] + '\n')
252
253        pos_cur = [i for i in range(len(fnct_src)) if re.search('tmp_from_original = 1', fnct_src[i])][0]
254        for ind, row in model_lhs_fa.iterrows():
255            fnct_src.insert(pos_cur, indnt + 'y = ' + row['token'] + '\n')
256
257    if model_lhs_fa1x.shape[0] > 0:
258
259        pos_cur = [i for i in range(len(fnct_src)) if re.search('tmp_to_original = 1', fnct_src[i])][0]
260        for ind, row in model_lhs_fa1x.iterrows():
261            fnct_src.insert(pos_cur, indnt + 'y = ' + row['token'] + '\n')
262
263    fn_src = [x for x in fnct_src if not re.search('tmp_(to|from)_original = 1', x)]
264    fn_cl = fun_generate(fn_src=fn_src
265                         , fn_name='transform_lhs_fn')
266
267    return fn_cl, fn_src

270def fun_find(fn_name: str
271             , env_code_src: dict | None):
272    """
273    Restores custom predict function (mostly category model).
274
275    Function cannot be pickled and restored properly w/o its environment so environment
276    to be loaded first.
277
278    :param fn_name: full function name including "path" (module tree with dots)
279    :param env_code_src: dictionary of source code of Python modules to look function through
280    :return:
281    """
282
283    if env_code_src is None or len(env_code_src) == 0:
284        fermatrica_error("Cannot restore function " + fn_name + ": adhoc code wasn't saved. " +
285                         "Try to re-save model properly")
286
287    for k, v in env_code_src.items():
288        fr_cur_name = inspect.currentframe().f_globals['__name__']
289        import_module_from_string(k, v, fr_cur_name)
290
291    fn = eval(fn_name)
292
293    return fn

296def fun_generate(fn_src: list | str
297                 , fn_name: str = 'transform_lhs_fn'):
298    """
299    Could generate any function from source, but mostly to generate transform_lhs_fn.
300
301    It is important to place it here and not in FERMATRICA_UTILS to add LHS functions
302    available in the environment. Another possible approach could be adding import
303    inside fun_generate as dynamic code from argument, but could be difficult and complicated
304    as well.
305
306    :param fn_src: function source code
307    :param fn_name: function name
308    :return:
309    """
310
311    d = {}
312
313    if not isinstance(fn_src, str):
314        fn_src = ''.join(fn_src)
315
316    exec(fn_src, globals(), d)
317
318    return d[fn_name]
